// A copied-in deep merge with no key filtering.
export default function tinyMerge(target: Record<string, any>, source: Record<string, any>) {
  (Reflect.ownKeys(source) as string[]).forEach((key) => {
    if (source[key] && typeof source[key] === 'object') tinyMerge(target[key], source[key]);
    else target[key] = source[key];
  });
  return target;
}
